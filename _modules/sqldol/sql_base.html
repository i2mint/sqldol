_SQL_IDENTIFIER_PART = r"(?=[\w$]*[^\W\d])[\w$]+"  # word chars or $, not all digits
SQL_IDENTIFIER_PATTERN = re.compile(
    rf"{_SQL_IDENTIFIER_PART}(\.{_SQL_IDENTIFIER_PART})?"
)
"""What a table name may look like where it is written into raw SQL text: letters
(Unicode included), digits, ``_`` or ``$``, not all digits (MySQL allows e.g.
``2020_sales``), optionally qualified by a schema (``schema.table``). None of these
characters can end an identifier or start a new SQL token."""

def validate_sql_identifier(name, *, pattern=SQL_IDENTIFIER_PATTERN):
    """Return ``name`` if it is safe to write into raw SQL text, else raise ``ValueError``.

    The raw-SQL paths of this module cannot bind a table name as a parameter (no SQL
    dialect allows that), and they may be handed a plain DB-API connection that has no
    quoting helper, so they only accept names matching an allowlist.

    >>> validate_sql_identifier("my_table")
    'my_table'
    >>> validate_sql_identifier("my_schema.my_table")
    'my_schema.my_table'
    >>> validate_sql_identifier("t; SELECT 1")  # doctest: +ELLIPSIS
    Traceback (most recent call last):
        ...
    ValueError: Not a valid SQL table name: 't; SELECT 1'. ...
    """
    if not isinstance(name, str) or not pattern.fullmatch(name):
        msg = (
            f"Not a valid SQL table name: {name!r}. "
            f"Expected letters, digits, '_' or '$' (not all digits), "
            f"optionally qualified as 'schema.table'. "
            f"Names that need quoting are not supported here: use the SQLAlchemy-based "
            f"stores (sqldol.base, sqldol.stores), which quote identifiers themselves."
        )
        raise ValueError(msg)
    return name

class SqlTableRowsCollection(Collection):
    """Base class wrapping an sql table.
    It is Iterable (yields rows (as tuples)) and Sized (i.e. you can call len on it).
    It's also a container, but brute-forced: should probably subclass if you want to perform `row in table` efficiently.

    Note: Be aware of how this object works if you're (1) talking to a table whose contents are changing dynamically.
    * count_rows() will return the current count every time
    * len returns the value of the _row_count lazyprop
    * _row_count returns the length of the _rows cache if it exists, and if not, will return the live count_rows
    * if len (therefore _row_count) is called before the object listed the rows (therefore filling it's cache),
        it will return the current number of rows, but if at the time of listing rows, the number of different,
        the number of rows will be updated to match number of rows cached.
    * column_names is a cached property
    """

    _tmpl_count_rows_tmpl = "SELECT COUNT(*) FROM {table_name}"
    _tmpl_describe_tmpl = "DESCRIBE {table_name}"
    _tmpl_iter_tmpl = "SELECT * FROM {table_name}"

    def __init__(self, connection, table_name, batch_size=2000, limit=int(1e16)):
        self.connection = connection
        # Note: table_name is written into raw SQL text (see the _tmpl_* templates),
        # so it has to pass the identifier allowlist.
        self.table_name = validate_sql_identifier(table_name)
        self.iter_rows = partial(
            iter_rows,
            connection=connection,
            table_name=table_name,
            batch_size=batch_size,
            limit=limit,
        )

    # Helpers ##########################################################################################################
    # QUESTION: Should helpers (_describe, _columns, etc.) should be methods/properties/lazyprops, and hidden or not?

    def count_rows(self):
        # Note: ``fetchone``, not ``first``: on a DB-API connection (the kind these
        # raw-SQL paths run on) ``execute`` returns a cursor, which has no ``first``.
        return self.connection.execute(
            self._tmpl_count_rows_tmpl.format(table_name=self.table_name)
        ).fetchone()[0]

    @lazyprop
    def _row_count(self):
        if lazyprop_w_sentinel.cache_is_active(self, "_rows"):
            return len(self._rows)
        else:
            return self.count_rows()

    def refresh_row_count(self):
        del self._row_count  # delete current
        return (
            self._row_count
        )  # return current row count (mainly to refresh the lazyprop

    def _describe(self):
        return self.connection.execute(
            self._tmpl_describe_tmpl.format(table_name=self.table_name)
        )

    @lazyprop
    def column_names(self):
        return tuple(x[0] for x in self._describe())

    @lazyprop_w_sentinel
    def _rows(self):
        rows = list(
            self.connection.execute(
                self._tmpl_iter_tmpl.format(table_name=self.table_name)
            )
        )
        self._row_count = len(rows)
        return rows

    ####################################################################################################################

    def __iter__(self):
        # Question: difference with `return iter(...)` for `for x in ...: yield x` options
        # Question: Why is the construction of the connection.execute slow for big tables (not a real cursor it seems)
        yield from self.iter_rows()

    def __len__(self):
        return self._row_count

    def __getitem__(self, idx):  # TODO: Make the ss[:-4] case work too
        if isinstance(idx, slice):
            start, stop, step = idx.start, idx.stop, idx.step
            start = start or 0
            assert step is None, "__getitem__ doesn't handle stepped slices"
            assert start >= 0, "slice start can't be negative"

            if stop is not None:
                assert stop >= start, "slice stop must be at least the slice start"
                return self.iter_rows(offset=start, limit=stop - start)
            elif start:
                return self.iter_rows(offset=start)
            else:
                return self.iter_rows()
        elif isinstance(idx, int):
            return self.iter_rows(offset=idx, limit=1)

    def __repr__(self):
        return f"SqlTable(..., table_name={self.table_name})"

class RowKey(dict):
    """A hashable, immutable ``{key_field: value}`` dict: the key of one row of a ``SQLAlchemyPersister``.

    It *is* a ``dict`` (so it works anywhere the persister takes a key -- ``store[key]``,
    ``key in store`` -- and equals the plain dict with the same items), but, unlike a plain
    dict, it is hashable, as a ``Mapping`` key must be (``set(store)``, ``dict(store)``,
    ``store.keys() & other``), and so cannot be mutated. Key fields can also be read as
    attributes (``key.id``), so code written for the ORM rows that ``__iter__`` used to yield
    (``row.id``) keeps working.

    >>> k = RowKey(id='a')
    >>> k == {'id': 'a'}, hash(k) == hash(RowKey(id='a')), {k: 1}[RowKey(id='a')], k.id
    (True, True, 1, 'a')
    >>> k['id'] = 'b'
    Traceback (most recent call last):
      ...
    TypeError: RowKey is immutable (it is hashable): build a new one
    """

    __slots__ = ()

    def __hash__(self):
        return hash(frozenset(self.items()))

    def __getattr__(self, name):
        try:
            return self[name]
        except KeyError:
            raise AttributeError(name) from None

    __setitem__ = __delitem__ = update = pop = popitem = clear = setdefault = _immutable
    __ior__ = _immutable

    def __reduce__(self):  # pickle/copy rebuild through __init__, not __setitem__
        return (RowKey, (dict(self),))




def _leaf_persister(store):
    """The ``SQLAlchemyPersister`` under any number of ``dol`` wrappers (``None`` if none)."""
    seen = set()
    while store is not None and id(store) not in seen:
        if isinstance(store, SQLAlchemyPersister):
            return store
        seen.add(id(store))
        store = getattr(store, "store", None)

_MAX_QUERY_PARAMS = 900  # stay under the oldest SQLite limit on bound variables


def _chunks(iterable, size):
    chunk = []
    for x in iterable:
        chunk.append(x)
        if len(chunk) >= size:
            yield chunk
            chunk = []
    if chunk:
        yield chunk


class _PrefetchingViewMixin:
    """Iterate ``(key, value)`` pairs of the caller's store a batch at a time, one query per batch.

    ``Mapping`` builds ``items()``/``values()`` from ``store[key]`` for every key, i.e. one
    query per row. Here the keys of a batch are resolved to the persister's own keys
    (``inner_most_key``), their rows are read by one query, and only while ``store[key]``
    runs are lookups of those keys answered from them. The lookups still go through the
    caller's own store, so key/value codecs and ``filt_iter`` scopes keep applying
    (i2mint/sqldol#11); nothing is cached while the caller's code runs between pairs.
    """

    def _pairs(self):
        mapping = self._mapping
        persister = _leaf_persister(mapping)
        if persister is None:
            for key in mapping:
                yield key, mapping[key]
            return
        for keys in _chunks(mapping, persister._prefetch_batch):
            rows = persister._load_rows(
                keys
                if mapping is persister
                else [inner_most_key(mapping, key, default=None) for key in keys]
            )
            for key in keys:
                with persister._using_rows(rows):
                    value = mapping[key]
                yield key, value


class _PrefetchingItemsView(_PrefetchingViewMixin, ItemsView):
    def __iter__(self):
        yield from self._pairs()


class _PrefetchingValuesView(_PrefetchingViewMixin, ValuesView):
    def __iter__(self):
        for _, value in self._pairs():
            yield value

class SQLAlchemyPersister(KvPersister):
    ItemsView = _PrefetchingItemsView
    ValuesView = _PrefetchingValuesView

    TYPE_INTEGER = sqlalchemy.INTEGER
    TYPE_STRING = sqlalchemy.String
    TYPE_BOOLEAN = sqlalchemy.BOOLEAN
    TYPE_BLOB = sqlalchemy.BLOB
    TYPE_TEXT = sqlalchemy.TEXT

    """
    A basic SQL DB persister written with SQLAlchemy.

    ``store.items()`` and ``store.values()`` (also through ``dol`` wrappers) read the table
    with one query per batch of keys (500) instead of one per key. Each batch is a snapshot
    taken when it starts; rows handed out stay in the session, as with any lookup. ``dict(store)`` does not (it asks ``keys()`` and
    then ``store[k]`` per key): use ``dict(store.items())``.

    ``table``, ``session`` and ``connection`` are the SQLAlchemy objects the persister is
    built on. ``dol`` wrappers serve them unchanged, so through a wrapper they are leaf
    escape hatches (the persister's coordinates, not the store's). The table's ``Query`` is
    ``sqldol.ops.query(store)``; ``query`` and ``setup`` are not methods any more
    (i2mint/sqldol#3).
    """

    def __init__(
        self,
        uri="sqlite:///my_sqlite.db",
        collection_name="dol_default_table",
        *,
        key_fields={"_id": TYPE_INTEGER},
        data_fields={"data": TYPE_STRING},
        autocommit=True,
        **db_kwargs,
    ):
        """
        :param uri: Uniform Resource Identifier of a database you would like to use.
            Unix/Mac (note the four leading slashes)
                sqlite:////absolute/path/to/foo.db

            Windows (note 3 leading forward slashes and backslash escapes)
                sqlite:///C:\\absolute\\path\\to\\foo.db

            Or go for in-memory DB with NO PERSISTANCE:
            sqlite:///:memory:

            Other options:
                postgresql://user:password@localhost:5432/my_db
                mysql://user:password@localhost/db
                oracle://user:password:tiger@localhost:1521/sidname

            I.e. in general:
                dialect+driver://username:password@host:port/database

        :param collection_name: name of the table to use, i.e. "my_table".
            If not given, it will be assumed that it's in the uri (i.e. the last part of the uri).
        :param key_fields: indexed keys columns names.
        :param data_fields: non-indexed data columns names.
        :param autocommit: whether each data change should be instantly commited, or not.
            It's off for context manager usecase (tbd).

        :param kwargs: any extra kwargs for SQLAlchemy engine to setup.
        """
        self._key_fields = key_fields
        self._data_fields = data_fields
        self.autocommit = autocommit

        self.connection = None
        self.table = None
        self.session = None

        # if collection_name is None:
        #     # TODO: Verify this logic and possibly refactor into a function
        #     uri, collection_name = _split_uri_into_base_and_collection_name(uri))

        self._uri = uri
        self._collection_name = collection_name
        self._setup(uri, collection_name, **db_kwargs)

    def setup(self, *args, **kwargs):
        """Removed: build a new store (``SQLAlchemyPersister(uri, collection_name, ...)``).

        A whole-store rebind cannot be intercepted or vetoed by the ``dol`` wrappers above
        the persister, which delegate public attributes to it, so under a wrapper it
        silently repointed the store at another database and table (i2mint/sqldol#3).
        """
        raise AttributeError(
            "SQLAlchemyPersister.setup was removed: it rebinds the whole store, below any "
            "dol wrapper. Build a new SQLAlchemyPersister (or store) for the other "
            "uri / collection_name instead (i2mint/sqldol#3)."
        )



    def _setup(self, db_uri, collection_name, **db_kwargs):
        self._uri = db_uri
        self._collection_name = collection_name
        # Setup connection to our DB:
        engine = create_engine(db_uri, **db_kwargs)
        self.connection = engine.connect()

        # Create a table (if not there yet)
        self.table = self._create_table(collection_name, engine)

        # Open ORM session:
        self.session = sessionmaker(bind=engine)()

    def _create_table(self, table_name, engine):
        """Create our data table (if not there yet)."""
        Base = declarative_base()

        table = Table(
            table_name,
            Base.metadata,
            *[
                Column(key, self._key_fields[key], primary_key=True, index=True)
                for key in self._key_fields
            ],
            *[Column(name, self._data_fields[name]) for name in self._data_fields],
        )
        table.create(bind=engine, checkfirst=True)

        # Lets wrap our Table with Base so we'll be able to use ORM features later.
        class DeclarativeTable(Base):
            __table__ = table

        return DeclarativeTable

    @property
    def _query(self):
        return self.session.query(self.table)

    @property
    def query(self):
        """Removed: use ``sqldol.ops.query(store)``.

        A public ``query`` is delegated by every ``dol`` wrapper, so ``SQLAlchemyTupleStore.query``
        handed back ORM rows while the store itself speaks tuples (i2mint/sqldol#3).
        """
        raise AttributeError(
            "SQLAlchemyPersister.query was removed: it answers in the persister's "
            "coordinates (ORM rows), whatever store wraps it. Use sqldol.ops.query(store) "
            "(i2mint/sqldol#3)."
        )

    _prefetch_batch = _PREFETCH_BATCH
    _prefetch = None  # {identity: row} while a view is looking up the rows it just read

    def _load_rows(self, keys):
        """``{identity: row}`` for the ``keys`` that name rows by exactly the key fields, by one query."""
        names = list(self._key_fields)
        idents = []
        for k in keys:
            if isinstance(k, Mapping) and set(k) == set(names):
                ident = tuple(k[name] for name in names)
                try:
                    hash(ident)
                except TypeError:  # unhashable value: looked up one by one
                    continue
                idents.append(ident)
        if not idents:
            return {}
        columns = [getattr(self.table, name) for name in names]
        # distinct, and in batches that keep the number of bound parameters small
        idents = list(dict.fromkeys(idents))
        rows = {}
        for part in _chunks(idents, max(1, _MAX_QUERY_PARAMS // len(names))):
            try:
                cond = (
                    columns[0].in_([i[0] for i in part])
                    if len(columns) == 1
                    else sqlalchemy.tuple_(*columns).in_(part)
                )
                for row in self._query.filter(cond):
                    rows[self._ident_of_row(row)] = row
            except (TypeError, sqlalchemy.exc.SQLAlchemyError):
                continue  # keys the database cannot compare: looked up one by one
        return rows

    @contextmanager
    def _using_rows(self, rows):
        previous, self._prefetch = self._prefetch, rows
        try:
            yield
        finally:
            self._prefetch = previous

    def _ident_of_row(self, row):
        return tuple(getattr(row, name) for name in self._key_fields)

    def _prefetched_row(self, k):
        """The prefetched row for key ``k`` (a mapping of exactly the key fields), or None.

        Only a row that is still persistent in the session under that identity, and not
        expired (a commit expires every row: its state in the database is unknown), counts:
        one deleted, detached, re-keyed or expired meanwhile is a miss and is looked up in
        the database, as a plain lookup does. Pending changes of the session are flushed
        first, as a plain lookup's query would.
        """
        if not self._prefetch or not isinstance(k, Mapping):
            return None
        if set(k) != set(self._key_fields):
            return None
        try:
            ident = tuple(k[name] for name in self._key_fields)
            row = self._prefetch.get(ident)
        except TypeError:  # unhashable key value
            return None
        if row is None:
            return None
        session = self.session
        if session.autoflush and (session.new or session.dirty or session.deleted):
            session.flush()
        state = sqlalchemy.inspect(row)
        if state.expired or state.expired_attributes:
            return None
        return row if state.persistent and state.identity == ident else None

    def __getitem__(self, k):
        doc = self._prefetched_row(k) or self._query.filter_by(**k).first()
        # todo: think of intuitive way of selecting many by 1 (or some) of many keys
        if not doc:
            raise KeyError(f"No document found for query: {k}")

        return doc

    def __contains__(self, k) -> bool:
        """True if and only if a row matches the key query ``k``.

        Getitem-based, so it agrees with ``__getitem__``, and it costs one
        query instead of the brute-force scan inherited from
        ``dol.base.Collection``. That inherited scan compared ``k`` against
        whatever ``__iter__`` yields -- ORM row objects, never keys -- and so
        answered False for every key, present or absent.

        A key that cannot name a row at all (not a mapping of field to value,
        or naming a field the table doesn't have) is simply not contained,
        which is what the inherited scan answered too.
        """
        try:
            return self._query.filter_by(**k).first() is not None
        except (TypeError, sqlalchemy.exc.InvalidRequestError):
            return False

    def __setitem__(self, k, v):
        try:
            doc = self[k]
        except KeyError:
            doc = self.table(**k, **v)
            self.session.add(doc)
        else:
            for key, value in v.items():
                setattr(doc, key, value)

        if self.autocommit:
            self.session.commit()

    def __delitem__(self, k):
        doc = self[k]
        self.session.delete(doc)

        if self.autocommit:
            self.session.commit()

    def __iter__(self):
        """Iterate the **keys** of the rows: ``RowKey`` dicts of ``{key_field: value}``.

        These are exactly what ``store[key]``, ``key in store`` and ``del store[key]`` take,
        so ``store[k] for k in store`` round-trips. (It used to yield ORM row objects, which
        no ``Mapping`` method accepts, i2mint/sqldol#2: use :meth:`iter_rows` for those.)
        """
        key_fields = list(self._key_fields)
        columns = [getattr(self.table, name) for name in key_fields]
        for values in self.session.query(*columns):
            yield RowKey(zip(key_fields, values))

def iter_rows(connection, table_name, batch_size=1000, offset=0, limit=int(1e12)):
    """Iterate over the rows of a table, fetching ``batch_size`` rows per query.

    Yields at most ``limit`` rows, starting at row ``offset`` (like SQL's
    ``LIMIT``/``OFFSET``), and stops at the end of the table: a page shorter than
    the one requested means there is nothing after it.

    ``table_name`` must pass :func:`validate_sql_identifier`, and ``batch_size``,
    ``offset`` and ``limit`` must be integers, because they are written into the SQL text
    (``batch_size`` positive, ``offset`` and ``limit`` non-negative). Invalid arguments
    raise ``ValueError`` at call time, before any row is requested.

    >>> import sqlite3
    >>> con = sqlite3.connect(":memory:")
    >>> _ = con.execute("CREATE TABLE t (x INTEGER)")
    >>> _ = con.executemany("INSERT INTO t VALUES (?)", [(i,) for i in range(5)])
    >>> list(iter_rows(con, "t", batch_size=2))
    [(0,), (1,), (2,), (3,), (4,)]
    >>> list(iter_rows(con, "t", batch_size=2, offset=1, limit=3))
    [(1,), (2,), (3,)]
    """
    table_name = validate_sql_identifier(table_name)
    batch_size, offset, limit = index(batch_size), index(offset), index(limit)
    if batch_size < 1:
        raise ValueError(f"batch_size must be a positive integer, got {batch_size}")
    if offset < 0 or limit < 0:
        raise ValueError(
            f"offset and limit must be non-negative, got offset={offset}, limit={limit}"
        )

    def pages():
        # Note: The end of the table is detected from the rows actually fetched,
        # never from ``rowcount``: for a SELECT that is -1 on DB-API drivers that
        # don't pre-buffer (sqlite3), so testing it made this loop request empty
        # pages until ``limit`` ran out.
        start, remaining = offset, limit
        while remaining > 0:
            page_size = min(batch_size, remaining)
            rows = connection.execute(
                f"SELECT * FROM {table_name} LIMIT {page_size} OFFSET {start}"
            ).fetchall()
            yield from rows
            if len(rows) < page_size:
                break
            remaining -= page_size
            start += page_size

    # Note: Not itself a generator, so bad arguments raise when ``iter_rows`` is
    # called; the rows are still fetched lazily, page by page.
    return pages()

class SQLAlchemyTupleStore(SQLAlchemyStore):
    @lazyprop
    def _key_fields(self):
        return self.store._key_fields

    def _id_of_key(self, k):
        return {field: field_val for field, field_val in zip(self._key_fields, k)}

    def _key_of_id(self, key):
        # ``key`` is a ``RowKey`` (what the persister iterates); an ORM row also works
        if isinstance(key, dict):
            return tuple(key[x] for x in self._key_fields)
        return tuple(getattr(key, x) for x in self._key_fields)

    @lazyprop
    def _data_fields(self):
        return self.store._data_fields

    def _data_of_obj(self, data):
        return {field: field_val for field, field_val in zip(self._data_fields, data)}

    def _obj_of_data(self, obj):
        return tuple(getattr(obj, x) for x in self._data_fields)
